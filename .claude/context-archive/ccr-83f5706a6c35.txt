     7	  const result: Record<string, unknown>[] = [];
     8	  for (const message of messages) {
     9	    if (message.role === "system") {
    10	      const sys = message as SystemMessage;
    11	      result.push({ role: "system", content: sys.content.map((c) => c.text).join("\n") });
    12	    } else if (message.role === "user") {
    13	      const user = message as UserMessage;
    14	      if (user.content.length === 1 && user.content[0]?.type === "text") {
    15	        result.push({ role: "user", content: user.content[0].text });
    16	      } else {
    17	        result.push({
    18	          role: "user",
    19	          content: user.content.map((c) => {
    20	            if (c.type === "text") return { type: "text", text: c.text };
    21	            if (c.type === "image_url") return { type: "image_url", image_url: c.image_url };
    22	            return null;
    23	          }).filter(Boolean),
    24	        });
    25	      }
    26	    } else if (message.role === "assistant") {
    27	      const asst = message as AssistantMessage;
    28	      const toolCalls = asst.content
    29	        .filter((c): c is ToolUseContent => c.type === "tool_use")
    30	        .map((c) => ({
    31	          type: "function" as const,
    32	          id: c.id,
    33	          function: { name: c.name, arguments: JSON.stringify(c.input) },
    34	        }));
    35	      const textParts = asst.content
    36	        .filter((c): c is TextContent => c.type === "text")
    37	        .map((c) => c.text)
    38	        .join("\n");
    39	      const entry: Record<string, unknown> = { role: "assistant" };
    40	      if (textParts) entry.content = textParts;
    41	      if (toolCalls.length > 0) entry.tool_calls = toolCalls;
    42	      if (!textParts && toolCalls.length === 0) entry.content = null;

    45	      const toolMsg = message as { role: "tool"; content: ToolResultContent[] };
    46	      for (const tr of toolMsg.content) {
    47	        result.push({ role: "tool", tool_call_id: tr.tool_use_id, content: tr.content });
